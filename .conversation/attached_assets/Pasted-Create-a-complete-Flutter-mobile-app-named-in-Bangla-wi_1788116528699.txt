CREATE TABLE munafa_settings (id INTEGER PRIMARY KEY, type TEXT UNIQUE, percent REAL)
INSERT INTO munafa_settings VALUES (1,'daily',5.0), (2,'weekly',7.0), (3,'monthly',10.0)

double dailyP=5, weeklyP=7, monthlyP=10 -> load from DB in initState

void recalculateMunafa(){
  double loan = double.tryParse(loanAmountController.text.replaceAll(',','').replaceAll('৳','').trim()) ?? 0;
  double percent = selectedCollectionType=='daily'? dailyP : selectedCollectionType=='weekly'? weeklyP : monthlyP;
  double munafa = loan * percent / 100;
  double total = loan + munafa;
  setState((){
    munafaController.text = munafa.toStringAsFixed(0);
    totalPayableController.text = total.toStringAsFixed(0);
  });
}
